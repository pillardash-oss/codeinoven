src/renderer/lib/app-typography.ts M 
// Font stacks for the family ids offered in Appearance settings.
export const FONT_STACKS: Record<string, string> = {
  'oven-mono': "'Oven Mono', ui-monospace, monospace",
  'oven-box': "'Oven Box', ui-monospace, monospace",
  'system': "-apple-system, sans-serif"
}

export function applyAppTypography(
  root: HTMLElement,
  typography: AppTypography
): void {
  const family = typography.fontFamily ?? 'oven-mono'
  root.style.setProperty('--font-app', FONT_STACKS[family])
  root.style.fontSize = `${typography.appFontSize}px`
}
